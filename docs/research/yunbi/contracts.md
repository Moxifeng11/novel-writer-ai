# 云笔数据契约（第二层交付）

> 来源：`/api/outlines/project/{id}`、`/api/chapters/project/{id}` 等真实响应（3.2MB）
> 提取日期：2026-11-26 · ZCode
> 用途：本仓可直接复用的 TypeScript 接口 + JSON 结构

## 一、核心洞察：章节大纲节点 = 36 字段「施工图」

云笔最值得学的一点：**章节大纲不是笔记，是给 AI 的完整施工图**。

对比：

| | 普通做法 | 云笔 |
| --- | --- | --- |
| 大纲内容 | 「主角拿到宝物，被追杀」 | 36 个字段：谁出场、在哪、说什么、情绪走向、埋什么伏笔、目标字数、开头承接、章尾钩子… |
| AI 拿到后 | 自由发挥 | 只能按施工图填 |

     * ```
     * - **【章首衔接】**：<与「开头承接」完全一致，独占首行>

     */

    人性细化: string;
    /** 本章核心台词（原文对话） */
    关键对话: string;
    /** 目标字数 */
    目标字数: number;
    /** 实际字数（写完回填） */
    实际字数: number;
    /** 正文字数（另一个口径，来自关联章节） */
    正文字数: number;

    // ── 实体白名单（写正文时的约束依据）──
    出场角色列表: YunbiCharacterRef[];
    场景列表: YunbiSceneRef[];
    道具物品: YunbiItemRef[];
    势力阵营: YunbiFactionRef[];

    // ── 情绪控制 ──
    情绪列表: YunbiEmotion[];
    本卷情绪控制: YunbiEmotion[];

    // ── 伏笔 ──
    关联伏笔列表: YunbiForeshadow[];
    本卷伏笔列表: YunbiForeshadow[];
    本卷大事件列表: Array<{ 名称: string; 影响: string; 描述: string }>;

    // ── 其他 ──
    身份关系: unknown[];
    出场章节: unknown[];
    参考素材: unknown[];
    时间线: unknown[];

### 2.3 协议行格式（CR/CI/NE/BT）

```ts
/** 协议行（正文末尾追加的结构化变更） */
export type YunbiProtocolKind = 'CR' | 'CI' | 'BT' | 'NE';

export interface YunbiProtocolLine {
    kind: YunbiProtocolKind;
    /** 按 `|` 分割的字段 */
    fields: string[];
}

/**
 * 四类协议行的字段约定：
 *
 * CR|关系方A|关系方B|本章关系变化|触发事件
 * CI|物品名|变动类型（获得/消耗/失去/损坏/赠与/转移）|数量
 * BT|角色名|原境界|新境界|突破契机
 * NE|名称|类型（角色/物品/地点/势力/功法）|为何必须出现
 */
export const YUNBI_PROTOCOL_SCHEMA: Record<YunbiProtocolKind, { name: string; fields: string[] }> = {
    CR: { name: '角色关系变化', fields: ['关系方A', '关系方B', '关系变化', '触发事件'] },
    CI: { name: '物品变动', fields: ['物品名', '变动类型', '数量'] },
    BT: { name: '境界突破', fields: ['角色名', '原境界', '新境界', '突破契机'] },
    NE: { name: '新实体', fields: ['名称', '类型', '理由'] },
};

/**
 * 从模型输出中分离正文与协议行。
 *
 * @param raw - 模型完整输出
 * @returns { body, lines }
 * @example
 * const { body, lines } = splitYunbiProtocol(raw);
 * // body 写正文，lines 交给结算层更新数据
 */
export function splitYunbiProtocol(raw: string): { body: string; lines: YunbiProtocolLine[] } {
    const lines: YunbiProtocolLine[] = [];
    const kept: string[] = [];
    for (const line of raw.split('\n')) {
        const t = line.trim();
        const m = /^(CR|CI|BT|NE)\|(.+)$/.exec(t);
        if (m) {
            lines.push({ kind: m[1] as YunbiProtocolKind, fields: m[2].split('|').map(x => x.trim()) });
        } else {
            kept.push(line);
        }
    }
    return { body: kept.join('\n').trimEnd(), lines };
}
```

### 2.4 时间线的数值排序编码（★ 值得抄的技术点）

```ts
/**
 * 云笔把 `纪年名-年-月-日` 编码成整数用于排序和链式校验。
 *
 * 例：`公元纪年-2-4-15` → `10002041500`
 * 拆解：1(纪年号) | 0002(年) | 04(月) | 15(日) | 00(时段)
 *
 * 好处：
 * - 整数比较代替日期字符串比较，避免「三月」vs「10月」的字典序错误
 * - 「第 N 章起始 ≥ 第 N-1 章结束」变成一次整数比较
 *
 * @param eraIndex - 纪年号（1 = 第一个历法）
 * @param year - 年
 * @param month - 月
 * @param day - 日
 * @param hourSlot - 时段（0-7，对应清晨~深夜）
 * @returns 可排序的整数
 * @example
 * encodeTimelineSortValue(1, 2, 4, 15, 0) // 10002041500
 */
export function encodeTimelineSortValue(
    eraIndex: number, year: number, month: number, day: number, hourSlot = 0,
): number {
    return eraIndex * 1e10 + year * 1e6 + month * 1e4 + day * 1e2 + hourSlot;
}

/** 时间线节点（18 字段） */
export interface YunbiTimelineNode {
    节点ID: number;
    项目ID: number;
    起始日期: string;
    结束日期: string;
    /** 数值排序值（见 encodeTimelineSortValue） */
    起始排序值: number;
    结束排序值: number;
    起始显示文本: string;
    结束显示文本: string;
    /**
     * 事件类型：
     * '主线情节' | '角色弧线' | '闪回' | '伏笔呼应' | '关键对话'
     * | '背景事件' | '时间跳跃' | '平行视角'
     */
    事件类型: string;
    所属轴: '主线' | '支线';
    绑定事件IDs: number[];
    主事件ID: number | null;
    起始时段: number;
    结束时段: number;
    /** 平行视角章的锚点节点（必须落在锚点章日期区间内） */
    并行锚点节点ID: number | null;
    校验状态: 'ok' | 'warn';
    校验信息: string;
    更新时间: string;
}
```

## 三、可以直接抄的设计原则

| # | 原则 | 为什么值得抄 |
| --- | --- | --- |
| 1 | **章节大纲是 36 字段施工图**，不是笔记 | AI 拿到后只能按图填，减少自由发挥 |
| 2 | **摘要用固定格式**（章首衔接→叙事→⟨✍批注⟩→伏笔→台词→章尾悬念） | 可解析、可校验、可精确定位注入 |
| 3 | **⟨✍⟩ 批注分离故事层/技法层** | 摘要说「发生什么」，批注说「怎么写」 |
| 4 | **实体白名单**（角色/场景/物品/势力各一表） | 防 AI 造新实体，且能精确注入 |
| 5 | **情绪列表描述「对读者的影响」**而非「角色的情绪」 | 这是写作技法，不是人物状态 |
| 6 | **伏笔带解密程度 + 预计回收章节** | 让伏笔有生命周期，不是一次性标记 |
| 7 | **时间线用整数编码排序** | 避免日期字符串比较的坑 |
| 8 | **情节脉络依赖前 9 步全部完成** | 强制依赖链，保证规划建立在完整设定上 |
| 9 | **协议行让模型顺手输出结构化变更** | 省一次 distill LLM 调用，且准确率更高 |
| 10 | **11 步每步带引导话术+输入占位+快速想法** | 降低填写门槛（用户只需回答「你想写什么」） |
